{{/* get-shows.html — single source of truth for parsing data/shows.yaml. Returns a dict: { "upcoming": [...], "past": [...] }, each show enriched with: ._at time.Time (parsed in Europe/Berlin, DST-aware) used for sorting/splitting ._atISO the same instant as an absolute ISO-8601 string — rendered into the markup so main.js can redo the split in the browser instead of leaving it frozen at build time ._hasTime bool — whether an explicit start time was given Used by the show-list shortcode (display) and schema.html (MusicEvent JSON-LD). */}} {{- $events := index hugo.Data.shows "shows" | default slice -}} {{- $now := now -}} {{- $upcoming := slice -}} {{- $past := slice -}} {{- range $e := $events -}} {{- $hasTime := and (isset $e "time") (ne (trim (string $e.time) " ") "") -}} {{- $raw := "" -}} {{- if $hasTime -}} {{- $raw = printf "%sT%s:00" (string $e.date) (string $e.time) -}} {{- else -}} {{- $raw = printf "%sT23:59:59" (string $e.date) -}} {{- end -}} {{- $at := time.AsTime $raw "Europe/Berlin" -}} {{- $m := merge $e (dict "_at" $at "_atISO" ($at.Format "2006-01-02T15:04:05-07:00") "_hasTime" $hasTime) -}} {{- if lt $at $now -}} {{- $past = $past | append $m -}} {{- else -}} {{- $upcoming = $upcoming | append $m -}} {{- end -}} {{- end -}} {{- $upcoming = sort $upcoming "_at" "asc" -}} {{- $past = sort $past "_at" "desc" -}} {{- return (dict "upcoming" $upcoming "past" $past) -}}
