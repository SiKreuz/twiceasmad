{{/* show-card.html — renders a single show card. Input: a show dict from get-shows.html. Optional merged-in flags: _isPast true → render the past variant _hidden true → render with the `hidden` attribute (teaser overflow) The card also carries everything main.js needs to turn an upcoming card into a past one in the browser once the show has happened: `data-show-at` (the moment it becomes past), `data-show-highlight` (teaser pinning), the "Vergangen" badge (pre-rendered but hidden) and `data-past-text` on the location fallback. */}} {{- $e := . -}} {{- $isPast := $e._isPast | default false -}} {{- $eventDate := $e._at -}} {{- $hasTime := $e._hasTime -}} {{- $mapsQuery := "" -}} {{- if and $e.venue $e.city -}} {{- $mapsQuery = printf "%s %s" $e.venue $e.city -}} {{- else if $e.venue -}} {{- $mapsQuery = $e.venue -}} {{- else if $e.city -}} {{- $mapsQuery = $e.city -}} {{- end -}}

{{ $eventDate.Format "02.01.2006" }} {{- if $hasTime }} {{ $e.time }} Uhr {{- end -}}

{{ $e.name }}

{{- if $mapsQuery -}} {{ with $e.venue }}{{ . }}{{ end }}{{ if and $e.venue $e.city }}, {{ end }}{{ with $e.city }}{{ . }}{{ end }} {{- else -}}

{{ if $isPast }}Ort unbekannt{{ else }}Ort folgt{{ end }}

{{- end -}}
